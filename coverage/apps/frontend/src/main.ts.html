import { AudioManager } from './audio/audio-manager';
import { ControlsUI } from './components/controls-ui';
import { drawSystem } from './canvas/renderer';
import { World } from '../../../packages/ecs/src/entities/world';
import { NutrientGrid } from '../../../packages/ecs/src/components/nutrient-grid';
import { TerrainGrid } from '../../../packages/ecs/src/components/terrain-grid';
import { behaviorSystem, decompositionSystem, disasterSystem, environmentSystem, interactionSystem, lifecycleSystem, movementSystem, seedWorld, spawnNutrientGrass, terrainSystem, DisasterState } from '../../../packages/ecs/src/systems/simulation-systems';
import { FIXED_STEP, SimulationConfig, Species } from '../../../packages/shared-types/src/ecs';
 
const canvas = document.querySelector<HTMLCanvasElement>('#biotope');
const status = document.querySelector<HTMLElement>('#status');
const controls = document.querySelector<HTMLElement>('#controls');
const audioButton = document.querySelector<HTMLButtonElement>('#audio-toggle');
if (!canvas || !status || !controls || !audioButton) throw new Error('Canvas initialization failed');
const context = canvas.getContext('2d');
if (!context) throw new Error('Canvas 2D context is unavailable');
const canvasElement = canvas;
const statusElement = status;
const renderingContext = context;
renderingContext.imageSmoothingEnabled = false;
 
const world = new World();
const nutrientGrid = new NutrientGrid(canvasElement.width, canvasElement.height);
const terrainGrid = new TerrainGrid(canvasElement.width, canvasElement.height);
const disasterState: DisasterState = { breedingRemaining: 0, meteorTriggered: false, floodTriggered: false };
const config: SimulationConfig = { width: canvasElement.width, height: canvasElement.height, initialGrass: 80, initialHerbivores: 18, initialCarnivores: 3, grassSpawnInterval: 1, maxGrass: 100, herbivoreSight: 5, splitEnergy: 150, carnivoreSpeed: 18, carnivoreMetabolism: 3.4 };
const controlsUI = new ControlsUI(controls, world, config, () => { nutrientGrid.clear(); terrainGrid.clear(); disasterState.breedingRemaining = 0; disasterState.meteorTriggered = false; disasterState.floodTriggered = false; });
const audio = new AudioManager(canvasElement.width);
const handleSimulationEvent = (event: Parameters<AudioManager['handleEvent']>[0]): void => audio.handleEvent(event);
document.addEventListener('pointerdown', () => audio.activate(), { once: true });
audioButton.addEventListener('click', () => { const enabled = audio.toggle(); audioButton.textContent = enabled ? '音響: ON' : '音響: OFF'; });
seedWorld(world, config);
let previousTime = performance.now();
let accumulator = 0;
let elapsed = 0;
let lastStatusTime = 0;
let frameCount = 0;
let fps = 0;
 
function tick(): void {
  const wetlandMultiplier = terrainGrid.wetlandArea() > 0 ? 10 : 1;
  const breedingMultiplier = disasterState.breedingRemaining > 0 ? 10 : 1;
  const tickConfig = { ...config, grassSpawnInterval: config.grassSpawnInterval / wetlandMultiplier / breedingMultiplier };
  spawnNutrientGrass(world, nutrientGrid, terrainGrid, elapsed, tickConfig, handleSimulationEvent);
  behaviorSystem(world, terrainGrid, config);
  environmentSystem(world, config.brightness ?? 1, config.temperature ?? 20);
  movementSystem(world, terrainGrid, FIXED_STEP, config.width, config.height);
  interactionSystem(world, terrainGrid, config, handleSimulationEvent);
  lifecycleSystem(world, elapsed, FIXED_STEP, config);
  decompositionSystem(world, nutrientGrid, terrainGrid, FIXED_STEP);
  terrainSystem(terrainGrid, handleSimulationEvent);
  disasterSystem(world, terrainGrid, disasterState, config.width, config.height, handleSimulationEvent);
  terrainGrid.advanceFlood(FIXED_STEP, config.height);
  world.commitCommands();
  elapsed += FIXED_STEP;
}
 
function frame(time: number): void {
  const delta = Math.min((time - previousTime) / 1000, 0.25);
  previousTime = time;
  accumulator += delta;
  while (accumulator >= FIXED_STEP) { tick(); accumulator -= FIXED_STEP; }
  drawSystem(world, nutrientGrid, terrainGrid, renderingContext);
  frameCount += 1;
  if (time - lastStatusTime > 250) {
    fps = frameCount * 1000 / (time - lastStatusTime || 1);
    statusElement.textContent = `FPS ${fps.toFixed(0)}  |  草 ${world.count(Species.Grass)}  草食 ${world.count(Species.Herbivore)}  肉食 ${world.count(Species.Carnivore)}`;
    audio.updateAmbient(world.count(Species.Grass), world.count(Species.Herbivore), world.count(Species.Carnivore));
    frameCount = 0;
    lastStatusTime = time;
  }
  requestAnimationFrame(frame);
}
