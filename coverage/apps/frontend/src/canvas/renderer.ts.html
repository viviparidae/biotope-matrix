import { NutrientGrid } from '../../../../packages/ecs/src/components/nutrient-grid';
import { TerrainGrid, TerrainKind, TERRAIN_CELL_SIZE } from '../../../../packages/ecs/src/components/terrain-grid';
import { World } from '../../../../packages/ecs/src/entities/world';
import { EntityFlags, MAX_ENTITIES, Species } from '../../../../packages/shared-types/src/ecs';
 
export function drawSystem(world: World, grid: NutrientGrid, terrain: TerrainGrid, context: CanvasRenderingContext2D): void {
	context.clearRect(0, 0, context.canvas.width, context.canvas.height);
	for (let row = 0; row < terrain.rows; row += 1) for (let column = 0; column < terrain.columns; column += 1) {
		const kind = terrain.kinds[row * terrain.columns + column] as TerrainKind;
		context.fillStyle = kind === TerrainKind.Desert ? '#b98b54' : kind === TerrainKind.Wetland ? '#2f7f83' : kind === TerrainKind.Mountain ? '#6b7280' : kind === TerrainKind.Crater ? '#342f38' : '#111827';
		context.fillRect(column * TERRAIN_CELL_SIZE, row * TERRAIN_CELL_SIZE, TERRAIN_CELL_SIZE, TERRAIN_CELL_SIZE);
	}
	if (terrain.waterFlow >= 0) {
		context.fillStyle = '#67e8f9';
		for (let column = 0; column < terrain.columns; column += 1) context.fillRect(column * TERRAIN_CELL_SIZE, Math.floor(terrain.waterFlow), TERRAIN_CELL_SIZE, 2);
	}
	for (let artifact = 0; artifact < world.artifactCount; artifact += 1) {
		const decomposing = world.artifactRemaining[artifact] > 0;
		context.fillStyle = decomposing ? '#58483d' : '#7a9b55';
		context.fillRect(Math.floor(world.artifactX[artifact]), Math.floor(world.artifactY[artifact]), 3, 3);
		if (!decomposing) {
			context.fillStyle = `rgba(150, 190, 92, ${Math.min(0.35, world.artifactResidual[artifact] / 50)})`;
			context.fillRect(Math.floor(world.artifactX[artifact]) - 3, Math.floor(world.artifactY[artifact]) - 3, 9, 9);
		}
	}
	for (let entity = 0; entity < MAX_ENTITIES; entity += 1) {
		if ((world.flags[entity] & EntityFlags.Alive) === 0) continue;
		const kind = world.species[entity];
		context.fillStyle = kind === Species.Grass ? '#65d66f' : kind === Species.Herbivore ? '#dbeafe' : '#ef4444';
		const size = kind === Species.Grass ? 1 : kind === Species.Herbivore ? 2 : 3;
		context.fillRect(Math.floor(world.x[entity]), Math.floor(world.y[entity]), size, size);
	}
	context.fillStyle = '#8fba62';
	for (let row = 0; row < grid.rows; row += 1) {
		for (let column = 0; column < grid.columns; column += 1) {
			if (grid.sample(column * TERRAIN_CELL_SIZE + 1, row * TERRAIN_CELL_SIZE + 1) > 0) context.fillRect(column * TERRAIN_CELL_SIZE, row * TERRAIN_CELL_SIZE, 1, 1);
		}
	}
}
